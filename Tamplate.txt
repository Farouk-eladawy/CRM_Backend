/**
 * الدالة الرئيسية - تعالج السجلات وترسل التفاصيل عبر الإيميل و Webhook
 */
function sendPickupDetails() {
  try {
    const records = fetchAirtableRecords();
    if (!records.length) {
      Logger.log('📭 لا توجد سجلات جديدة للمعالجة.');
      return;
    }

    Logger.log(`🔄 جاري معالجة ${records.length} سجل...`);
    
    let successCount = 0;
    let errorCount = 0;
    let emailsSentInBatch = 0;

    records.forEach((record, index) => {
      try {
        // إضافة تأخير بين كل إيميل
        if (index > 0) {
          Utilities.sleep(EMAIL_DELAY_MS);
        }
        
        // إضافة تأخير إضافي بعد كل دفعة
        if (emailsSentInBatch >= BATCH_SIZE) {
          Logger.log(`⏸️ الانتظار ${BATCH_DELAY_MS / 1000} ثانية بعد إرسال ${BATCH_SIZE} إيميلات...`);
          Utilities.sleep(BATCH_DELAY_MS);
          emailsSentInBatch = 0;
        }
        
        const result = processRecord(record);
        if (result.emailSent) {
          emailsSentInBatch++;
        }
        successCount++;
      } catch (error) {
        errorCount++;
        Logger.log(`❌ خطأ أثناء معالجة السجل ${record.id}: ${error.message}`);
        
        // إذا كان الخطأ بسبب تجاوز الحد، توقف عن المعالجة
        if (error.message.includes('User-rate limit exceeded')) {
          Logger.log('⚠️ تم الوصول إلى حد معدل الإرسال. سيتم إيقاف المعالجة.');
          return;
        }
      }
    });

    Logger.log(`✅ اكتملت المعالجة. نجاح: ${successCount}, فشل: ${errorCount}`);
  } catch (error) {
    Logger.log(`❌ خطأ عام: ${error.message}`);
  }
}

/**
 * التحقق من كون الرحلة رحلة طيران
 * @param {String} tripName - اسم الرحلة
 * @return {Boolean} - true إذا كانت رحلة طيران
 */
function isFlightTrip(tripName) {
  const tripNameLower = tripName.toLowerCase();
  return tripNameLower.includes('plane') || tripNameLower.includes('flight');
}

/**
 * التحقق من كون الرحلة رحلة شاطئ تحتاج ملاحظة خاصة
 * @param {String} tripName - اسم الرحلة
 * @return {Boolean} - true إذا كانت رحلة شاطئ
 */
function isBeachTrip(tripName) {
  const tripNameLower = tripName.toLowerCase();
  return BEACH_TRIPS.some(beachTrip => tripNameLower.includes(beachTrip));
}

/**
 * تحديد نوع الإيميل الأساسي
 * @param {String} tripName - اسم الرحلة
 * @return {String} - نوع الإيميل (ticket, pickup)
 */
function determineEmailType(tripName) {
  const tripNameLower = tripName.toLowerCase();
  
  // إيميل التذاكر للرحلات التي تحتوي على QR أو Sound
  if (tripNameLower.includes('qr') || tripNameLower.includes('sound')) {
    return 'ticket';
  }
  
  // إيميل تفاصيل الاستلام العادي (بما في ذلك رحلات الطيران)
  return 'pickup';
}

/**
 * معالجة سجل واحد من البيانات
 * @param {Object} record - سجل البيانات من Airtable
 * @return {Object} - نتيجة المعالجة
 */
function processRecord(record) {
  const fields = record.fields;
  Logger.log(`📎 Attachments field type: ${Object.prototype.toString.call(fields['Attachments'])}`);
  
  // استخراج البيانات وتنظيفها
  const bookingData = extractBookingData(fields);
  
  // تحديد نوع الإيميل الأساسي
  const emailType = determineEmailType(bookingData.tripName);
  const isFlightBooking = isFlightTrip(bookingData.tripName);
  Logger.log(`📧 نوع الإيميل المحدد: ${emailType} للرحلة: ${bookingData.tripName}${isFlightBooking ? ' (رحلة طيران - سيتم إرسال رسالة مطار إضافية)' : ''}`);

  // تجهيز محتوى الإيميل الأساسي
  const emailSubject = generateEmailSubject(bookingData);
  let emailBody = '';
  
  switch (emailType) {
    case 'ticket':
      emailBody = generateTicketEmail(bookingData);
      break;
    default:
      emailBody = generatePickupDetailsEmail(bookingData);
  }
  
  // تجهيز بيانات الـ Webhook
  const webhookPayload = prepareWebhookPayload(record.id, bookingData);
  
  // إرسال الإيميل الأساسي والتعامل مع الأخطاء
  let emailSent = false;
  let airportEmailSent = false;
  
  if (bookingData.customerEmail && isValidEmail(bookingData.customerEmail)) {
    try {
      // إرسال الإيميل الأساسي (pickup أو ticket)
      sendEmailWithRetry(bookingData.customerEmail, emailSubject, emailBody);
      Logger.log(`📧 تم إرسال إيميل ${emailType} إلى: ${bookingData.customerEmail}`);
      emailSent = true;
      
      // إرسال رسالة المطار الإضافية للرحلات الجوية
      if (isFlightBooking) {
        try {
          // إضافة تأخير قبل إرسال الإيميل الثاني
          Utilities.sleep(EMAIL_DELAY_MS);
          
          const airportSubject = `Important Airport Instructions - ${bookingData.tripName} ${bookingData.bookingNr}`;
          const airportBody = generateAirportEmail(bookingData);
          
          sendEmailWithRetry(bookingData.customerEmail, airportSubject, airportBody);
          Logger.log(`✈️ تم إرسال رسالة المطار الإضافية إلى: ${bookingData.customerEmail}`);
          airportEmailSent = true;
        } catch (airportError) {
          Logger.log(`❌ فشل إرسال رسالة المطار: ${airportError.message}`);
          sendErrorAlert(bookingData, `فشل إرسال رسالة المطار: ${airportError.message}`);
        }
      }
      
    } catch (emailError) {
      Logger.log(`❌ فشل إرسال الإيميل الأساسي: ${emailError.message}`);
      sendErrorAlert(bookingData, `فشل إرسال الإيميل الأساسي: ${emailError.message}`);
      throw emailError; // إعادة رمي الخطأ للتعامل معه في الدالة الرئيسية
    }
  } else if (bookingData.customerEmail) {
    // إرسال تنبيه عن إيميل غير صالح
    sendErrorAlert(bookingData, 'الإيميل غير صالح');
  }
  
  // إرسال Webhook
  const webhookSent = sendWebhook(webhookPayload);
  
  // تحديد حالة الإرسال
  const overallEmailSuccess = emailSent && (!isFlightBooking || airportEmailSent);
  
  // تحديث حالة السجل
  const status = (overallEmailSuccess || !bookingData.customerEmail) && webhookSent ? 'Done' : 'Try Again';
  updateWhatsappStatus(record.id, status);
  
  Logger.log(`✓ تمت معالجة السجل ${record.id}: إيميل أساسي=${emailSent}${isFlightBooking ? `, إيميل مطار=${airportEmailSent}` : ''}, webhook=${webhookSent}, الحالة=${status}`);
  
  return { emailSent: emailSent || airportEmailSent };
}

/**
 * إرسال إيميل مع إعادة المحاولة في حالة الفشل
 * @param {String} to - البريد الإلكتروني للمستلم
 * @param {String} subject - عنوان الإيميل
 * @param {String} htmlBody - محتوى الإيميل HTML
 */
function sendEmailWithRetry(to, subject, htmlBody, maxRetries = 3) {
  let lastError;
  
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      GmailApp.sendEmail(to, subject, '', {
        htmlBody: htmlBody,
        name: COMPANY_NAME
      });
      return; // نجح الإرسال
    } catch (error) {
      lastError = error;
      
      if (error.message.includes('User-rate limit exceeded')) {
        // استخراج وقت إعادة المحاولة من رسالة الخطأ
        const retryMatch = error.message.match(/Retry after (.+?) \(/);
        if (retryMatch) {
          const retryTime = new Date(retryMatch[1]);
          const waitTime = Math.max(0, retryTime - new Date()) + 1000; // إضافة ثانية إضافية
          
          if (attempt < maxRetries) {
            Logger.log(`⏳ الانتظار ${Math.ceil(waitTime / 1000)} ثانية قبل إعادة المحاولة ${attempt + 1}/${maxRetries}...`);
            Utilities.sleep(waitTime);
          }
        } else if (attempt < maxRetries) {
          // إذا لم نتمكن من استخراج وقت إعادة المحاولة، انتظر وقت افتراضي
          const defaultWait = 5000 * attempt; // 5، 10، 15 ثانية
          Logger.log(`⏳ الانتظار ${defaultWait / 1000} ثانية قبل إعادة المحاولة ${attempt + 1}/${maxRetries}...`);
          Utilities.sleep(defaultWait);
        }
      } else {
        // خطأ آخر غير متعلق بمعدل الإرسال
        throw error;
      }
    }
  }
  
  // فشلت جميع المحاولات
  throw lastError;
}

/**
 * استخراج وتنظيف بيانات الحجز من حقول السجل
 * @param {Object} fields - حقول السجل من Airtable
 * @return {Object} - بيانات الحجز المنظفة
 */
function extractBookingData(fields) {
  const attachmentsArray = fields['Attachments'] || [];

  return {
    bookingNr: cleanField(fields['Booking Nr.']),
    dateTrip: formatDate(fields['Date Trip']),
    tripName: cleanField(fields['trip Name']),
    optionName: cleanField(fields['Option']),
    customerName: formatCustomerName(cleanField(fields['Customer Name'])),
    customerEmail: cleanField(fields['Customer Email']),
    customerPhone: cleanPhone(fields['Customer Phone']),
    hotelName: cleanField(fields['Hotel Name']),
    pickupTime: cleanField(fields['pickup time']),
    addOns: cleanField(fields['Collecting on date Trip']),
    nonBillableAddons: cleanField(fields['Non-billable addons']),
    googleMaps: cleanField(fields['Google Maps']),
    attachments: extractAttachmentsHTML(attachmentsArray), // لعرض المرفقات في الإيميل
    attachmentsArray: attachmentsArray,                    // لإرسالها في Webhook
    fullTripName: prepareFullTripName(
      cleanField(fields['trip Name']),
      cleanField(fields['Option'])
    )
  };
}

/**
 * تجهيز Payload لـ Webhook بعد تنظيف البيانات
 * @param {String} recordId - معرف السجل
 * @param {Object} data - بيانات الحجز
 * @return {Object} - Payload جاهز للإرسال
 */
function prepareWebhookPayload(recordId, data) {
  const attachmentsForWebhook = extractAttachmentsForWebhook(data.attachmentsArray);

  const payload = {
    recordId: recordId,
    bookingNr: data.bookingNr,
    dateTrip: data.dateTrip,
    tripName: data.tripName,
    optionName: data.optionName,
    customerName: data.customerName,
    customerEmail: data.customerEmail,
    customerPhone: data.customerPhone,
    hotelName: data.hotelName,
    pickupTime: data.pickupTime,
    addOns: data.addOns,
    nonBillableAddons: data.nonBillableAddons,
    googleMaps: data.googleMaps,
    ...attachmentsForWebhook
  };

  return cleanPayload(payload);
}

/**
 * جلب السجلات من Airtable
 * @return {Array} - مصفوفة من السجلات
 */
function fetchAirtableRecords() {
  try {
    const response = UrlFetchApp.fetch(
      `https://api.airtable.com/v0/${BASE_ID}/${TABLE_NAME}?view=${encodeURIComponent(VIEW_NAME)}`, 
      { method: 'get', headers: HEADERS, muteHttpExceptions: true }
    );
    
    const statusCode = response.getResponseCode();
    if (statusCode !== 200) {
      throw new Error(`فشل جلب السجلات. كود الاستجابة: ${statusCode}`);
    }
    
    const content = JSON.parse(response.getContentText());
    return content.records || [];
  } catch (error) {
    Logger.log(`❌ خطأ في جلب السجلات: ${error.message}`);
    return [];
  }
}

/**
 * إرسال البيانات إلى Webhook
 * @param {Object} payload - البيانات المراد إرسالها
 * @return {Boolean} - نجاح العملية
 */
function sendWebhook(payload) {
  try {
    const options = {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    };
    
    const response = UrlFetchApp.fetch(WEBHOOK_URL, options);
    const statusCode = response.getResponseCode();
    
    if (statusCode !== 200) {
      Logger.log(`⚠️ استجابة Webhook غير متوقعة: ${statusCode}`);
    }
    
    return statusCode === 200;
  } catch (error) {
    Logger.log(`❌ فشل إرسال Webhook: ${error.message}`);
    return false;
  }
}

/**
 * التحقق من صحة الإيميل
 * @param {String} email - البريد الإلكتروني للتحقق
 * @return {Boolean} - صحة البريد الإلكتروني
 */
function isValidEmail(email) {
  const pattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return pattern.test(email);
}

/**
 * تنظيف حقل نصي
 * @param {String} value - القيمة المراد تنظيفها
 * @return {String} - القيمة بعد التنظيف
 */
function cleanField(value) {
  return (value || '')
    .toString()
    .replace(/[\r\n]+/g, ' ')      // إزالة أي أسطر جديدة وتحويلها إلى مسافة واحدة
    .replace(/\([^\)]*\)/g, '')    // حذف أي محتوى داخل أقواس ()
    .replace(/,+/g, ',')           // دمج الفواصل المكررة إلى فاصلة واحدة فقط
    .replace(/\s*,\s*/g, ',')      // حذف المسافات حول الفواصل
    .replace(/\s+/g, ' ')          // دمج كل المسافات الزائدة إلى مسافة واحدة
    .trim();
}

/**
 * تنظيف نص وتحويل المسافات إلى فواصل
 * @param {String} text - النص المراد تنظيفه
 * @return {String} - النص بعد التنظيف
 */
function cleanText(text) {
  return (text || '').toString().trim().replace(/\s+/g, ',');
}

/**
 * تنظيف رقم الهاتف من المسافات
 * @param {String} phone - رقم الهاتف المراد تنظيفه
 * @return {String} - رقم الهاتف بعد التنظيف
 */
function cleanPhone(phone) {
  return (phone || '').toString().trim().replace(/\s+/g, '');
}

/**
 * تنسيق التاريخ بصيغة dd/MM/yyyy
 * @param {String} dateStr - سلسلة التاريخ
 * @return {String} - التاريخ المنسق
 */
function formatDate(dateStr) {
  if (!dateStr) return '';
  try {
    const date = new Date(dateStr);
    return Utilities.formatDate(date, Session.getScriptTimeZone(), 'dd/MM/yyyy');
  } catch (error) {
    return dateStr || '';
  }
}

/**
 * تجهيز اسم الرحلة الكامل
 * @param {String} tripName - اسم الرحلة
 * @param {String} optionName - اسم الخيار
 * @return {String} - اسم الرحلة الكامل
 */
function prepareFullTripName(tripName, optionName) {
  if (optionName) {
    return `${tripName} - ${optionName}`;
  }
  return tripName;
}

/**
 * تنسيق اسم العميل
 * @param {String} name - اسم العميل
 * @return {String} - اسم العميل المنسق
 */
function formatCustomerName(name) {
  const parts = cleanField(name).split(' ');
  if (parts.length > 1) {
    return parts.slice(0, 2).join(' ') + ',' + (parts.length > 2 ? ' ' + parts.slice(2).join(' ') : '');
  }
  return name;
}

/**
 * حذف الحقول الفارغة من الـ Payload
 * @param {Object} payload - الكائن المراد تنظيفه
 * @return {Object} - الكائن بعد التنظيف
 */
function cleanPayload(payload) {
  const cleaned = {};
  for (const key in payload) {
    if (payload[key] !== undefined && 
        payload[key] !== null && 
        payload[key].toString().trim() !== '') {
      cleaned[key] = payload[key];
    }
  }
  return cleaned;
}

/**
 * تجهيز HTML للمرفقات في الإيميل
 * @param {Array} attachments - مصفوفة المرفقات
 * @return {String} - HTML المرفقات
 */
function extractAttachmentsHTML(attachments) {
  if (!attachments || !attachments.length) return '';

  return attachments.map((att, idx) => {
    const driveUrl = uploadToDriveAndGetPublicUrl(att.url, att.filename || `Attachment_${idx + 1}.pdf`);
    return `<p><b>${att.filename || 'Attachment ' + (idx + 1)}</b>: 
      <a href="${driveUrl}" target="_blank" style="background-color: ${EMAIL_COLORS.secondary}; color: white; padding: 5px 10px; text-decoration: none; border-radius: 4px;">View Ticket</a></p>`;
  }).join('');
}

/**
 * تجهيز معلومات المرفقات للـ Webhook
 * @param {Array} attachments - مصفوفة المرفقات
 * @return {Object} - كائن يحتوي على معلومات المرفقات
 */
function extractAttachmentsForWebhook(attachments) {
  let result = {};
  if (attachments && attachments.forEach) {
    attachments.forEach((att, idx) => {
      result[`Attachment ${idx + 1} Name`] = att.filename || `Attachment ${idx + 1}`;
      result[`Attachment ${idx + 1} Download URL`] = att.url || '';
    });
  }
  return result;
}
