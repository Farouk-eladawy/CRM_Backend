from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.chrome.options import Options
import json
import time
import re
from datetime import datetime

class GetYourGuideScraperFixed:
    def __init__(self, cookies=None):
        chrome_options = Options()
        chrome_options.add_argument('--start-maximized')
        chrome_options.add_argument('--disable-blink-features=AutomationControlled')
        
        self.driver = webdriver.Chrome(options=chrome_options)
        self.wait = WebDriverWait(self.driver, 20)
        self.cookies = cookies
        
    def login_with_cookies(self):
        if self.cookies:
            self.driver.get('https://supplier.getyourguide.com/')
            time.sleep(2)
            for cookie in self.cookies:
                try:
                    self.driver.add_cookie(cookie)
                except:
                    pass
            time.sleep(1)
    
    def extract_from_full_text(self, full_text):
        """استخراج البيانات من الـ full_text مباشرة"""
        data = {}
        
        # Product ID
        match = re.search(r'Product Id:\s*(\d+)', full_text)
        data['product_id'] = match.group(1) if match else None
        
        # Reference Code
        match = re.search(r'Product Reference Code:\s*(\S+)', full_text)
        data['reference_code'] = match.group(1) if match else None
        
        # Rating
        match = re.search(r'Rating:\s*([\d.]+)\s*of\s*5', full_text)
        data['rating'] = match.group(1) if match else None
        
        # Short Description
        match = re.search(r'Short description\s*\n\s*(.+?)(?:\n\nFull description|$)', full_text, re.DOTALL)
        data['short_description'] = match.group(1).strip() if match else None
        
        # Full Description
        match = re.search(r'Full description\s*\n(.+?)(?:\nSee less\nHighlights|Highlights)', full_text, re.DOTALL)
        if match:
            desc = match.group(1).strip()
            # نظف النص من "See less"
            desc = re.sub(r'\nSee less.*$', '', desc, flags=re.DOTALL)
            data['full_description'] = desc
        else:
            data['full_description'] = None
        
        # Highlights
        highlights = []
        match = re.search(r'Highlights\s*\n(.+?)(?:\nSee less|Inclusions & Exclusions)', full_text, re.DOTALL)
        if match:
            highlight_text = match.group(1)
            # استخرج كل السطور اللي مش فارغة
            for line in highlight_text.split('\n'):
                line = line.strip()
                if line and not line.startswith('See ') and line != 'See less':
                    # نظف من الرصاصات
                    line = re.sub(r'^[•\-]\s*', '', line)
                    if len(line) > 3:  # تجنب السطور القصيرة جداً
                        highlights.append(line)
        data['highlights'] = highlights[:10]  # أول 10
        
        # Inclusions
        inclusions = []
        match = re.search(r'Inclusions\s*\n(.+?)(?:\nSee less|Exclusions)', full_text, re.DOTALL)
        if match:
            inclusion_text = match.group(1)
            for line in inclusion_text.split('\n'):
                line = line.strip()
                if line and not line.startswith('See '):
                    line = re.sub(r'^[•\-]\s*', '', line)
                    if len(line) > 3:
                        inclusions.append(line)
        data['inclusions'] = inclusions[:20]
        
        # Exclusions
        exclusions = []
        match = re.search(r'Exclusions\s*\n(.+?)(?:\nSee less|Important information)', full_text, re.DOTALL)
        if match:
            exclusion_text = match.group(1)
            if 'No exclusions added' not in exclusion_text:
                for line in exclusion_text.split('\n'):
                    line = line.strip()
                    if line and not line.startswith('See '):
                        line = re.sub(r'^[•\-]\s*', '', line)
                        if len(line) > 3:
                            exclusions.append(line)
        data['exclusions'] = exclusions[:20]
        
        # Keywords
        keywords = []
        match = re.search(r'Keywords\s*\n(.+?)(?:\nSee all|Guide Information|Transportation)', full_text, re.DOTALL)
        if match:
            keyword_text = match.group(1)
            # استخرج الكلمات من سطر واحد أو عدة سطور
            words = re.findall(r'\b[A-Z][a-z]+(?:\s+[a-z]+)*\b', keyword_text)
            for word in words:
                if word not in ['See', 'all', 'keywords'] and len(word) > 2:
                    keywords.append(word)
        data['keywords'] = list(set(keywords))[:20]
        
        # Transportation
        match = re.search(r'Transportation\s*\n(.+?)(?:\nRefund policy|Food & Drinks)', full_text, re.DOTALL)
        if match:
            transport = match.group(1).strip()
            # نظف من السطور الزايدة
            transport = re.sub(r'\n+', ', ', transport)
            transport = transport.replace('No transportation provided for this product', '')
            data['transportation'] = transport.strip() if transport.strip() else None
        else:
            data['transportation'] = None
        
        # Refund Policy
        match = re.search(r'Refund policy\s*\n(.+?)(?:\nItinerary|History)', full_text, re.DOTALL)
        if match:
            refund = match.group(1).strip()
            # خد أول سطر بس
            refund = refund.split('\n')[0]
            data['refund_policy'] = refund if refund else None
        else:
            data['refund_policy'] = None
        
        # Food & Drinks
        food_drinks = {}
        match = re.search(r'Food & Drinks\s*\nMeal\s*\n\s*Format\s*\n(.+?)(?:\nKeywords|$)', full_text, re.DOTALL)
        if match:
            food_text = match.group(1).strip()
            lines = [l.strip() for l in food_text.split('\n') if l.strip()]
            if len(lines) >= 2 and 'No food or drinks' not in food_text:
                food_drinks['meal'] = lines[0].replace('\t', ' ')
                food_drinks['format'] = lines[1] if len(lines) > 1 else None
        data['food_drinks'] = food_drinks
        
        # Important Information
        important_info = {}
        
        # Emergency Contact
        match = re.search(r'Emergency contact number\s*\n({[^}]+})', full_text)
        if match:
            important_info['emergency_contact'] = match.group(1)
        
        # Pet Policy
        match = re.search(r'Pet policy\s*\n(.+?)(?:\nWhat mandatory|$)', full_text, re.DOTALL)
        if match:
            pet = match.group(1).strip().split('\n')[0]
            if 'No restrictions' not in pet:
                important_info['pet_policy'] = pet
        
        # Know before you go
        match = re.search(r'Know before you go\s*\n(.+?)(?:\nSee less|Emergency contact)', full_text, re.DOTALL)
        if match:
            know = match.group(1).strip()
            if 'No extra information' not in know:
                # قسّم لنقاط
                points = [p.strip() for p in know.split('\n') if p.strip() and len(p.strip()) > 10]
                if points:
                    important_info['know_before_you_go'] = points[:10]
        
        data['important_information'] = important_info
        
        return data
    
    def scrape_tour(self, tour_id):
        url = f'https://supplier.getyourguide.com/products/details?tour_id={tour_id}'
        
        try:
            print(f"Scraping tour {tour_id}...")
            self.driver.get(url)
            time.sleep(4)  # انتظر تحميل الصفحة
            
            # احصل على full_text
            full_text = self.driver.find_element(By.TAG_NAME, 'body').text
            
            # احصل على title من الـ h1
            try:
                title = self.driver.find_element(By.TAG_NAME, 'h1').text
            except:
                # استخرج من النص
                match = re.search(r'^(.+?)\n(?:No active|Rejected|Product Id)', full_text, re.MULTILINE)
                title = match.group(1).strip() if match else "Supply Partner |"
            
            # استخرج البيانات من النص
            extracted_data = self.extract_from_full_text(full_text)
            
            # بناء الـ tour_data
            tour_data = {
                'url': url,
                'scraped_at': datetime.now().strftime('%Y-%m-%d %H:%M:%S'),
                'title': title,
                'product_id': extracted_data.get('product_id'),
                'reference_code': extracted_data.get('reference_code'),
                'rating': extracted_data.get('rating', ''),
                'short_description': extracted_data.get('short_description'),
                'full_description': extracted_data.get('full_description'),
                'highlights': extracted_data.get('highlights', []),
                'inclusions': extracted_data.get('inclusions', []),
                'exclusions': extracted_data.get('exclusions', []),
                'important_information': extracted_data.get('important_information', {}),
                'options': self.extract_options(full_text),
                'keywords': extracted_data.get('keywords', []),
                'transportation': extracted_data.get('transportation'),
                'refund_policy': extracted_data.get('refund_policy'),
                'food_drinks': extracted_data.get('food_drinks', {}),
                'full_text': full_text
            }
            
            print(f"✓ Tour {tour_id} scraped successfully")
            print(f"  - Product ID: {tour_data['product_id']}")
            print(f"  - Reference: {tour_data['reference_code']}")
            print(f"  - Highlights: {len(tour_data['highlights'])}")
            print(f"  - Options: {len(tour_data['options'])}")
            
            return tour_data
            
        except Exception as e:
            print(f"✗ Error scraping tour {tour_id}: {e}")
            return None
    
    def extract_options(self, full_text):
        """استخراج الـ options من النص"""
        options = []
        
        # ابحث عن كل option
        option_pattern = r'Title\s*\n(.+?)\s*\nReference code\s*\n(.+?)\s*\nOption ID\s*\n(\d+)\s*\nStatus\s*\n(Active|Deactivated)(?:.*?\nType\s*\n(Private|Non-private))?'
        
        matches = re.finditer(option_pattern, full_text, re.DOTALL)
        
        for match in matches:
            option = {
                'title': match.group(1).strip(),
                'reference_code': match.group(2).strip(),
                'option_id': match.group(3).strip(),
                'status': match.group(4).strip(),
                'type': match.group(5).strip() if match.group(5) else None
            }
            options.append(option)
        
        return options
    
    def scrape_multiple_tours(self, tour_ids, output_file='tours_scraped.json'):
        results = []
        
        for i, tour_id in enumerate(tour_ids, 1):
            print(f"\n[{i}/{len(tour_ids)}] Processing tour {tour_id}")
            print("="*60)
            
            tour_data = self.scrape_tour(tour_id)
            
            if tour_data:
                results.append(tour_data)
                
                # احفظ بعد كل tour
                with open(output_file, 'w', encoding='utf-8') as f:
                    json.dump(results, f, ensure_ascii=False, indent=2)
                print(f"✓ Progress saved ({len(results)} tours)")
            
            # Delay
            if i < len(tour_ids):
                time.sleep(3)
        
        return results
    
    def close(self):
        self.driver.quit()

def fix_existing_json(input_file='tours_data.json', output_file='tours_fixed.json'):
    """
    إصلاح البيانات الناقصة من ملف JSON موجود
    """
    with open(input_file, 'r', encoding='utf-8') as f:
        tours = json.load(f)
    
    scraper = GetYourGuideScraperFixed()
    
    try:
        # حمّل cookies
        try:
            with open('getyourguide_cookies.json', 'r') as f:
                cookies = json.load(f)
                scraper.cookies = cookies
                scraper.login_with_cookies()
        except:
            print("No cookies - continuing without login")
        
        fixed_tours = []
        
        for i, tour in enumerate(tours, 1):
            print(f"\n[{i}/{len(tours)}] Fixing tour data...")
            
            # استخرج tour_id من URL
            match = re.search(r'tour_id=(\d+)', tour['url'])
            if not match:
                print("  ✗ Could not extract tour ID")
                fixed_tours.append(tour)
                continue
            
            tour_id = match.group(1)
            
            # أعد scrape إذا البيانات ناقصة
            if not tour.get('product_id') or not tour.get('reference_code'):
                print(f"  Re-scraping tour {tour_id}...")
                new_data = scraper.scrape_tour(tour_id)
                if new_data:
                    fixed_tours.append(new_data)
                else:
                    fixed_tours.append(tour)
            else:
                # استخدم البيانات من full_text إذا موجودة
                if tour.get('full_text'):
                    print(f"  Extracting from full_text for tour {tour_id}...")
                    extracted = scraper.extract_from_full_text(tour['full_text'])
                    
                    # حدّث البيانات الناقصة فقط
                    if not tour.get('product_id'):
                        tour['product_id'] = extracted.get('product_id')
                    if not tour.get('reference_code'):
                        tour['reference_code'] = extracted.get('reference_code')
                    if not tour.get('short_description'):
                        tour['short_description'] = extracted.get('short_description')
                    if not tour.get('full_description'):
                        tour['full_description'] = extracted.get('full_description')
                    if not tour.get('highlights'):
                        tour['highlights'] = extracted.get('highlights', [])
                    if not tour.get('inclusions'):
                        tour['inclusions'] = extracted.get('inclusions', [])
                    if not tour.get('keywords'):
                        tour['keywords'] = extracted.get('keywords', [])
                    
                    print(f"  ✓ Fixed: ID={tour['product_id']}, Ref={tour['reference_code']}")
                
                fixed_tours.append(tour)
            
            time.sleep(2)
        
        # احفظ النتائج
        with open(output_file, 'w', encoding='utf-8') as f:
            json.dump(fixed_tours, f, ensure_ascii=False, indent=2)
        
        print(f"\n✓ Fixed data saved to {output_file}")
        
        return fixed_tours
        
    finally:
        scraper.close()

def quick_scrape(tour_ids_or_file):
    """
    Scrape بسرعة - فقط أدخل IDs أو ملف
    
    Args:
        tour_ids_or_file: قائمة بالـ IDs أو مسار لملف JSON
    """
    # حدد tour_ids
    if isinstance(tour_ids_or_file, str) and tour_ids_or_file.endswith('.json'):
        # اقرأ من JSON
        with open(tour_ids_or_file, 'r') as f:
            data = json.load(f)
        tour_ids = []
        for item in data:
            match = re.search(r'tour_id=(\d+)', item['url'])
            if match:
                tour_ids.append(match.group(1))
    else:
        tour_ids = tour_ids_or_file
    
    print(f"🚀 Starting scraper for {len(tour_ids)} tours...")
    
    # Scrape
    scraper = GetYourGuideScraperFixed()
    
    try:
        results = scraper.scrape_multiple_tours(tour_ids, 'tours_complete.json')
        
        print(f"\n{'='*60}")
        print(f"✅ DONE! Scraped {len(results)}/{len(tour_ids)} tours")
        print(f"📁 Saved to: tours_complete.json")
        print(f"{'='*60}")
        
        # إحصائيات
        successful = sum(1 for t in results if t.get('product_id'))
        print(f"\n📊 Statistics:")
        print(f"  Complete data: {successful}/{len(results)}")
        print(f"  Avg highlights: {sum(len(t['highlights']) for t in results)/len(results):.1f}")
        print(f"  Avg inclusions: {sum(len(t['inclusions']) for t in results)/len(results):.1f}")
        
        return results
        
    finally:
        scraper.close()

def extract_data_from_json_file(json_file):
    """
    استخرج البيانات الناقصة من JSON موجود باستخدام full_text
    بدون الحاجة لفتح المتصفح!
    """
    with open(json_file, 'r', encoding='utf-8') as f:
        tours = json.load(f)
    
    scraper = GetYourGuideScraperFixed()
    
    fixed_count = 0
    
    for tour in tours:
        if tour.get('full_text'):
            print(f"\nProcessing: {tour['url']}")
            
            # استخرج من النص
            extracted = scraper.extract_from_full_text(tour['full_text'])
            
            # حدّث البيانات
            updated = False
            
            if not tour.get('product_id') and extracted.get('product_id'):
                tour['product_id'] = extracted['product_id']
                updated = True
            
            if not tour.get('reference_code') and extracted.get('reference_code'):
                tour['reference_code'] = extracted['reference_code']
                updated = True
            
            if not tour.get('short_description') and extracted.get('short_description'):
                tour['short_description'] = extracted['short_description']
                updated = True
            
            if not tour.get('full_description') and extracted.get('full_description'):
                tour['full_description'] = extracted['full_description']
                updated = True
            
            if not tour.get('highlights') and extracted.get('highlights'):
                tour['highlights'] = extracted['highlights']
                updated = True
            
            if not tour.get('inclusions') and extracted.get('inclusions'):
                tour['inclusions'] = extracted['inclusions']
                updated = True
            
            if not tour.get('keywords') and extracted.get('keywords'):
                tour['keywords'] = extracted['keywords']
                updated = True
            
            if not tour.get('transportation') and extracted.get('transportation'):
                tour['transportation'] = extracted['transportation']
                updated = True
            
            if updated:
                fixed_count += 1
                print(f"  ✓ Fixed: ID={tour['product_id']}, Ref={tour['reference_code']}")
    
    # احفظ
    output_file = json_file.replace('.json', '_fixed.json')
    with open(output_file, 'w', encoding='utf-8') as f:
        json.dump(tours, f, ensure_ascii=False, indent=2)
    
    print(f"\n✅ Fixed {fixed_count} tours")
    print(f"📁 Saved to: {output_file}")
    
    return tours
