/**
 * الدالة الرئيسية - تعالج السجلات وترسل التفاصيل عبر الإيميل و Webhook
 */
function sendPickupDetails() {
  try {
    const records = fetchAirtableRecords();
    if (!records.length) {
      Logger.log('📭 لا توجد سجلات جديدة للمعالجة.');
      return;
    }

    Logger.log(`🔄 جاري معالجة ${records.length} سجل...`);
    
    let successCount = 0;
    let errorCount = 0;

    records.forEach(record => {
      try {
        processRecord(record);
        successCount++;
      } catch (error) {
        errorCount++;
        Logger.log(`❌ خطأ أثناء معالجة السجل ${record.id}: ${error.message}`);
      }
    });

    Logger.log(`✅ اكتملت المعالجة. نجاح: ${successCount}, فشل: ${errorCount}`);
  } catch (error) {
    Logger.log(`❌ خطأ عام: ${error.message}`);
  }
}
  /**

 * @param {String} url - رابط الملف الأصلي

 */
function uploadToDriveAndGetPublicUrl(url, filename) {
  try {
    const response = UrlFetchApp.fetch(url);

    const folder = DriveApp.getFolderById("19kxt_vaqX3ZuvfxmuR7gO9ffpRc2-kXb"); // ضع ID المجلد هنا

}

/**
 * معالجة سجل واحد من البيانات
 * @param {Object} record - سجل البيانات من Airtable
 */
function processRecord(record) {
  const fields = record.fields;
  Logger.log(`📎 Attachments field type: ${Object.prototype.toString.call(fields['Attachments'])}`);
  
  // استخراج البيانات وتنظيفها
  const bookingData = extractBookingData(fields);
  
  // تحديد نوع الإيميل المناسب
  const isTicketEmail = bookingData.tripName.includes('QR') || bookingData.tripName.includes('Sound');
  

  // تجهيز محتوى الإيميل
  const emailSubject = generateEmailSubject(bookingData);
  const emailBody = isTicketEmail ? 
    generateTicketEmail(bookingData) : 
    generatePickupDetailsEmail(bookingData);
  
  // تجهيز بيانات الـ Webhook
  const webhookPayload = prepareWebhookPayload(record.id, bookingData);
  
  // إرسال الإيميل والتعامل مع الأخطاء
  let emailSent = false;
  
  if (bookingData.customerEmail && isValidEmail(bookingData.customerEmail)) {
    try {
      GmailApp.sendEmail(bookingData.customerEmail, emailSubject, '', {
        htmlBody: emailBody,
        name: COMPANY_NAME
      });
      Logger.log(`📧 تم إرسال الإيميل إلى: ${bookingData.customerEmail}`);
      emailSent = true;
    } catch (emailError) {
      Logger.log(`❌ فشل إرسال الإيميل: ${emailError.message}`);
      sendErrorAlert(bookingData, `فشل إرسال الإيميل: ${emailError.message}`);
    }
  } else if (bookingData.customerEmail) {
    // إرسال تنبيه عن إيميل غير صالح
    sendErrorAlert(bookingData, 'الإيميل غير صالح');
  }
  
  // إرسال Webhook
  const webhookSent = sendWebhook(webhookPayload);
  
  // تحديث حالة السجل
  const status = (emailSent || !bookingData.customerEmail) && webhookSent ? 'Done' : 'Try Again';
  updateWhatsappStatus(record.id, status);
  
  Logger.log(`✓ تمت معالجة السجل ${record.id}: إيميل=${emailSent}, webhook=${webhookSent}, الحالة=${status}`);
}

/**
 * استخراج وتنظيف بيانات الحجز من حقول السجل
 * @param {Object} fields - حقول السجل من Airtable
 * @return {Object} - بيانات الحجز المنظفة
 */
function extractBookingData(fields) {
  const allAttachments = fields['Attachments'] || [];
  
  // فلترة المرفقات: تجاهل الصور (QR Codes) واستخدام ملفات PDF فقط (أو أي ملفات أخرى غير الصور)
  const attachmentsArray = allAttachments.filter(att => {
    const isImage = att.type ? att.type.startsWith('image/') : 
                   (att.filename.toLowerCase().endsWith('.png') || 
                    att.filename.toLowerCase().endsWith('.jpg') || 
                    att.filename.toLowerCase().endsWith('.jpeg'));
    return !isImage;
  });

  // معالجة المرفقات (رفعها إلى Drive للحصول على روابط دائمة)
  const processedAttachments = attachmentsArray.map((att, idx) => {
    let filename = att.filename || `Ticket_${idx + 1}`;
    // التأكد من أن اسم الملف ينتهي بـ .pdf
    if (!filename.toLowerCase().endsWith('.pdf')) {
      filename += '.pdf';
    }

    // نرفع الملف إلى Google Drive ونحصل على الرابط الدائم
    const driveUrl = uploadToDriveAndGetPublicUrl(att.url, filename);
    
    return {
      filename: filename,
      url: driveUrl, // الرابط الدائم من Drive
      type: 'application/pdf' // تحديد النوع كـ PDF بشكل صريح
    };
  });

  return {
    bookingNr: cleanField(fields['Booking Nr.']),
    dateTrip: formatDate(fields['Date Trip']),
    tripName: cleanField(fields['trip Name']),
    optionName: cleanField(fields['Option']),
    customerName: formatCustomerName(cleanField(fields['Customer Name'])),
    customerEmail: cleanField(fields['Customer Email']),
    customerPhone: cleanPhone(fields['Customer Phone']),
    hotelName: cleanField(fields['Hotel Name']),
    pickupTime: cleanField(fields['pickup time']),
    addOns: cleanField(fields['Collecting on date Trip']),
    nonBillableAddons: cleanField(fields['Non-billable addons']),
    googleMaps: cleanField(fields['Google Maps']),
    attachments: generateAttachmentsHTML(processedAttachments), // استخدام المرفقات المعالجة
    attachmentsArray: processedAttachments,                     // إرسال المرفقات المعالجة (بروابط Drive) للـ Webhook
    fullTripName: prepareFullTripName(
      cleanField(fields['trip Name']),
      cleanField(fields['Option'])
    )
  };
}

}

/**
 * جلب السجلات من Airtable
 * @return {Array} - مصفوفة من السجلات
 */
function fetchAirtableRecords() {
  try {
    const response = UrlFetchApp.fetch(
      `https://api.airtable.com/v0/${BASE_ID}/${TABLE_NAME}?view=${encodeURIComponent(VIEW_NAME)}`, 
      { method: 'get', headers: HEADERS, muteHttpExceptions: true }
    );
    
    const statusCode = response.getResponseCode();
    if (statusCode !== 200) {
      throw new Error(`فشل جلب السجلات. كود الاستجابة: ${statusCode}`);
    }
    
    const content = JSON.parse(response.getContentText());
    return content.records || [];
  } catch (error) {
    Logger.log(`❌ خطأ في جلب السجلات: ${error.message}`);
    return [];
  }
}

/**
 * إرسال البيانات إلى Webhook
 * @param {Object} payload - البيانات المراد إرسالها
 * @return {Boolean} - نجاح العملية
 */
function sendWebhook(payload) {
  try {
    const options = {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    };
    
    const response = UrlFetchApp.fetch(WEBHOOK_URL, options);
    const statusCode = response.getResponseCode();
    
    if (statusCode !== 200) {
      Logger.log(`⚠️ استجابة Webhook غير متوقعة: ${statusCode}`);
    }
    
    return statusCode === 200;
  } catch (error) {
    Logger.log(`❌ فشل إرسال Webhook: ${error.message}`);
    return false;
  }
}

/**
 * التحقق من صحة الإيميل
 * @param {String} email - البريد الإلكتروني للتحقق
 * @return {Boolean} - صحة البريد الإلكتروني
 */
function isValidEmail(email) {
  const pattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return pattern.test(email);
}

/**
 * تنظيف حقل نصي
 * @param {String} value - القيمة المراد تنظيفها
 * @return {String} - القيمة بعد التنظيف
 */
function cleanField(value) {
  return (value || '')
    .toString()
    .replace(/[\r\n]+/g, ' ')      // إزالة أي أسطر جديدة وتحويلها إلى مسافة واحدة
    .replace(/\([^\)]*\)/g, '')    // حذف أي محتوى داخل أقواس ()
    .replace(/,+/g, ',')           // دمج الفواصل المكررة إلى فاصلة واحدة فقط
    .replace(/\s*,\s*/g, ',')      // حذف المسافات حول الفواصل
    .replace(/\s+/g, ' ')          // دمج كل المسافات الزائدة إلى مسافة واحدة
    .trim();
}

/**
 * تنظيف نص وتحويل المسافات إلى فواصل
 * @param {String} text - النص المراد تنظيفه
 * @return {String} - النص بعد التنظيف
 */
function cleanText(text) {
  return (text || '').toString().trim().replace(/\s+/g, ',');
}

/**
 * تنظيف رقم الهاتف من المسافات
 * @param {String} phone - رقم الهاتف المراد تنظيفه
 * @return {String} - رقم الهاتف بعد التنظيف
 */
function cleanPhone(phone) {
  return (phone || '').toString().trim().replace(/\s+/g, '');
}

/**
 * تنسيق التاريخ بصيغة dd/MM/yyyy
 * @param {String} dateStr - سلسلة التاريخ
 * @return {String} - التاريخ المنسق
 */
function formatDate(dateStr) {
  if (!dateStr) return '';
  try {
    const date = new Date(dateStr);
    return Utilities.formatDate(date, Session.getScriptTimeZone(), 'dd/MM/yyyy');
  } catch (error) {
    return dateStr || '';
  }
}

/**
 * تجهيز اسم الرحلة الكامل
 * @param {String} tripName - اسم الرحلة
 * @param {String} optionName - اسم الخيار
 * @return {String} - اسم الرحلة الكامل
 */
function prepareFullTripName(tripName, optionName) {
  if (optionName) {
    return `${tripName} - ${optionName}`;
  }
  return tripName;
}

/**
 * تنسيق اسم العميل
 * @param {String} name - اسم العميل
 * @return {String} - اسم العميل المنسق
 */
function formatCustomerName(name) {
  const parts = cleanField(name).split(' ');
  if (parts.length > 1) {
    return parts.slice(0, 2).join(' ') + ',' + (parts.length > 2 ? ' ' + parts.slice(2).join(' ') : '');
  }
  return name;
}

/**
 * حذف الحقول الفارغة من الـ Payload
 * @param {Object} payload - الكائن المراد تنظيفه
 * @return {Object} - الكائن بعد التنظيف
 */
function cleanPayload(payload) {
  const cleaned = {};
  for (const key in payload) {
    if (payload[key] !== undefined && 
        payload[key] !== null && 
        payload[key].toString().trim() !== '') {
      cleaned[key] = payload[key];
    }
  }
  return cleaned;
}
