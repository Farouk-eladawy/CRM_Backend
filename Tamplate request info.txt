/**
 * Main function - Entry point for the script
 * Gets records and processes each one
 */
function sendThanksEmails() {
  // Fetch records that haven't received welcome emails yet
  const filterFormula = encodeURIComponent('NOT({Welcome email sent?})');
  const records = fetchRecordsFromAirtable(filterFormula);
  
  if (!records.length) {
    Logger.log('📭 No new bookings found requiring welcome emails.');
    return;
  }
  
  Logger.log(`🔍 Found ${records.length} new bookings to process.`);
  
  // Process each record
  let successCount = 0;
  let errorCount = 0;
  
  records.forEach(record => {
    try {
      const result = processBooking(record);
      if (result) successCount++;
    } catch (error) {
      errorCount++;
      Logger.log(`❌ Error processing booking (${record.id}): ${error.message}`);
    }
  });
  
  Logger.log(`✅ Process completed. Success: ${successCount}, Errors: ${errorCount}`);
}

/**
 * Process a single booking record
 * @param {Object} record - The Airtable record to process
 * @returns {boolean} Success status
 */
function processBooking(record) {
  const fields = record.fields;
  
  // Extract and clean essential fields
  const bookingData = {
    recordId: record.id,
    customerEmail: cleanField(fields['Customer Email']),
    tripName: cleanField(fields['trip Name']),
    bookingNr: cleanField(fields['Booking Nr.']),
    customerName: formatCustomerName(cleanField(fields['Customer Name'])),
    optionName: cleanField(fields['Option']),
    agency: cleanField(fields['Agency']), // ✅ إضافة حقل Agency
    destination: cleanField(fields['des']) // ✅ إضافة حقل الوجهة
  };
  
  // Validate email
  if (!bookingData.customerEmail || !isValidEmail(bookingData.customerEmail)) {
    Logger.log(`⚠️ Skipping booking due to invalid email: Booking Nr. ${bookingData.bookingNr}, Email: ${bookingData.customerEmail}`);
    return false;
  }
  
  // Create full trip name with option if available
  const fullTripName = bookingData.optionName ? 
    `${bookingData.tripName} (${bookingData.optionName})` : 
    bookingData.tripName;
  
  // ✅ تحديث شرط إرسال إيميل التذاكر ليشمل Agency = Tiqets
  let emailSent = false;
  if (bookingData.tripName.includes('QR') || 
      bookingData.tripName.includes('Sound') || 
      bookingData.agency === 'Tiqets') {
    emailSent = sendTicketEmail(bookingData, fullTripName);
  } else {
    emailSent = sendBookingConfirmationEmail(bookingData, fullTripName);
  }
  
  // Update record status if email was sent successfully
  if (emailSent) {
    updateWelcomeEmailStatus(bookingData.recordId);
    return true;
  }
  
  return false;
}

/**
 * Send ticket-type email for QR code, Sound & Light shows, or Tiqets bookings
 * @param {Object} bookingData - The booking information
 * @param {string} fullTripName - Complete trip name with options
 * @returns {boolean} Success status
 */
function sendTicketEmail(bookingData, fullTripName) {
  const subject = `Your Tickets for ${fullTripName} - ${bookingData.bookingNr} ✨`;
  // ✅ تمرير معلومة Agency إلى دالة generateTicketEmail
  const body = generateTicketEmail(bookingData.customerName, fullTripName, bookingData.bookingNr, bookingData.agency);
  
  try {
    GmailApp.sendEmail(bookingData.customerEmail, subject, '', {
      htmlBody: body,
      name: CONFIG.email.senderName
    });
    
    Logger.log(`🎟️ Ticket email sent to: ${bookingData.customerEmail} (Booking: ${bookingData.bookingNr})`);
    return true;
  } catch (error) {
    Logger.log(`❌ Failed to send ticket email: ${error.message}`);
    return false;
  }
}

/**
 * Fetch records from Airtable with optional filter
 * @param {string} filterFormula - Optional Airtable filter formula
 * @returns {Array} Array of record objects
 */
function fetchRecordsFromAirtable(filterFormula = '') {
  const baseUrl = `https://api.airtable.com/v0/${CONFIG.airtable.baseId}/${CONFIG.airtable.tableName}?view=${CONFIG.airtable.viewName}`;
  const url = filterFormula ? `${baseUrl}&filterByFormula=${filterFormula}` : baseUrl;
  
  try {
    const response = UrlFetchApp.fetch(url, {
      method: 'get',
      headers: HEADERS,
      muteHttpExceptions: true
    });
    
    const responseCode = response.getResponseCode();
    
    if (responseCode !== 200) {
      Logger.log(`⚠️ Airtable API returned status ${responseCode}`);
      return [];
    }
    
    const json = JSON.parse(response.getContentText());
    return json.records || [];
  } catch (error) {
    Logger.log(`❌ Failed to fetch records: ${error.message}`);
    return [];
  }
}

/**
 * Update the "Welcome email sent?" field in Airtable
 * @param {string} recordId - The Airtable record ID
 * @returns {boolean} Success status
 */
function updateWelcomeEmailStatus(recordId) {
  const url = `https://api.airtable.com/v0/${CONFIG.airtable.baseId}/${CONFIG.airtable.tableName}/${recordId}`;
  const payload = JSON.stringify({
    fields: {
      "Welcome email sent?": true
    }
  });
  
  try {
    const response = UrlFetchApp.fetch(url, {
      method: 'patch',
      headers: HEADERS,
      payload: payload,
      muteHttpExceptions: true
    });
    
    const responseCode = response.getResponseCode();
    if (responseCode === 200) {
      Logger.log(`✅ Updated record status: ${recordId}`);
      return true;
    } else {
      Logger.log(`⚠️ Failed to update record status. Response code: ${responseCode}`);
      return false;
    }
  } catch (error) {
    Logger.log(`❌ Error updating record status: ${error.message}`);
    return false;
  }
}

/**
 * Validate email format
 * @param {string} email - Email address to validate
 * @returns {boolean} Is valid email
 */
function isValidEmail(email) {
  const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return emailPattern.test(email);
}

/**
 * Clean and format text fields
 * @param {string} value - Raw text value
 * @returns {string} Cleaned text
 */
function cleanField(value) {
  return (value || '')
    .toString()
    .replace(/[\r\n]+/g, ' ')      // Replace newlines with spaces
    .replace(/\([^\)]*\)/g, '')    // Remove content in parentheses
    .replace(/,+/g, ',')           // Merge multiple commas
    .replace(/\s*,\s*/g, ',')      // Remove spaces around commas
    .replace(/\s+/g, ' ')          // Merge multiple spaces
    .trim();
}

/**
 * Format customer name nicely
 * @param {string} name - Raw customer name
 * @returns {string} Formatted name
 */
function formatCustomerName(name) {
  const cleanedName = cleanField(name);
  const parts = cleanedName.split(' ');
  
  if (parts.length > 1) {
    // Format name with first two parts as first name, rest as last name
    return parts.slice(0, 2).join(' ') + (parts.length > 2 ? ', ' + parts.slice(2).join(' ') : '');
  }
  
  return cleanedName;
}
